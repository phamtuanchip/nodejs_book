Chương 14: Caching với Redis — Tăng Tốc Ứng Dụng

Redis là in-memory database nhanh nhất thế giới. Dùng làm cache giúp response time giảm từ 200ms xuống 2ms — gấp 100 lần. Chương này học Redis từ cơ bản đến các patterns thực tế: cache-aside, session store, rate limiting, và pub/sub.

🎯 Sau chương này bạn sẽ
  • Hiểu Redis là gì và kiến trúc in-memory database
  • Làm chủ 5 data types: String, Hash, List, Set, Sorted Set
  • Implement Cache-Aside pattern — pattern phổ biến nhất
  • Viết Cache middleware tự động cho Express
  • Thay thế MemoryStore bằng Redis Session Store
  • Xây dựng Rate Limiter bằng Redis từ scratch
  • Pub/Sub cơ bản cho real-time notifications
  • Distributed lock để tránh race conditions

14.1 Redis là gì?

Redis (Remote Dictionary Server) là in-memory key-value store — nghĩa là nó lưu toàn bộ data trong RAM thay vì ổ cứng. Đây là lý do nó nhanh đến mức khó tin: RAM nhanh hơn SSD tới 1000 lần.

Storage Tốc độ đọc Latency điển hình
HDD (ổ cứng cũ) ~200 MB/s 5–10ms
SSD NVMe ~3500 MB/s 0.05–0.1ms
RAM (Redis) ~50,000 MB/s <0.1ms

Tại sao Redis không mất data khi tắt máy?

Redis có 2 cơ chế persistence:

  • RDB (Redis Database): Snapshot — chụp toàn bộ data ra file .rdb định kỳ (mỗi 5 phút). Nếu server crash, mất tối đa vài phút data.
  • AOF (Append-Only File): Ghi từng command vào log file. Khi restart, replay lại log. An toàn hơn nhưng tốn dung lượng hơn.

Redis dùng cho gì?

  • Cache: Lưu kết quả query DB, API responses tốn thời gian
  • Session Store: Lưu user sessions cho app multi-server
  • Rate Limiting: Giới hạn số request/phút từ 1 IP
  • Pub/Sub: Message broker đơn giản giữa các service
  • Leaderboard: Sorted set cho ranking real-time
  • Job Queue: FIFO queue cho background jobs (Bull/BullMQ dùng Redis)
  • Distributed Lock: Đồng bộ hoá giữa nhiều server instances

14.3 Data Types — 5 Kiểu Dữ Liệu Cần Biết

String — Kiểu phổ biến nhất

String trong Redis có thể chứa text, numbers, hoặc binary data (JSON, serialized objects).

JavaScriptString operations
const redis = require('./lib/redis');

// ── SET với TTL (Time To Live) ──
await redis.set('greeting', 'Hello World');            // Không expire
await redis.set('user:42:name', 'Alice', 'EX', 3600); // Expire sau 1 giờ (seconds)
await redis.set('token:abc123', userId, 'PX', 300000); // Expire sau 5 phút (milliseconds)

// isSet = 'OK' nếu set thành công, null nếu key đã tồn tại

// ── GET ──
const name = await redis.get('user:42:name'); // 'Alice' hoặc null nếu không tồn tại
const json = await redis.get('product:1');    // String, cần JSON.parse

// ── DEL ──
await redis.del('user:42:name');
await redis.del(['key1', 'key2', 'key3']); // Xóa nhiều cùng lúc

// ── EXISTS ──
const exists = await redis.exists('user:42:name'); // 1 nếu có, 0 nếu không

// ── TTL — xem thời gian sống còn lại ──
const ttl = await redis.ttl('user:42:name'); // Số giây còn lại, -1 nếu không expire, -2 nếu không tồn tại
await redis.expire('user:42:name', 7200);    // Đặt TTL mới (2 giờ)

// ── Numeric operations ──
await redis.set('page:views', 0);
await redis.incr('page:views');          // Tăng 1 → atomic!
await redis.incrby('page:views', 5);     // Tăng 5
await redis.decr('page:views');          // Giảm 1
await redis.incrbyfloat('price:btc', 100.5); // Float increment

// ── MSET / MGET — bulk operations ──
await redis.mset('key1', 'val1', 'key2', 'val2', 'key3', 'val3');
const [v1, v2, v3] = await redis.mget('key1', 'key2', 'key3'); // [val1, val2, val3]

// ── ZSCORE — điểm của 1 user ──
const score = await redis.zscore('leaderboard:global', 'alice'); // '1600'

// ── ZRANGE + WITHSCORES — toàn bộ bảng xếp hạng ──
const all = await redis.zrange('leaderboard:global', 0, -1, 'WITHSCORES');

// ── ZRANGEBYSCORE — lấy users trong khoảng score ──
const midRange = await redis.zrangebyscore('leaderboard:global', 1000, 2000, 'WITHSCORES');

// Leaderboard helper function
async function getLeaderboard(page = 1, pageSize = 10) {
  const start = (page - 1) * pageSize;
  const end = start + pageSize - 1;

  const raw = await redis.zrevrange('leaderboard:global', start, end, 'WITHSCORES');

  // Parse kết quả: ['alice', '1600', 'bob', '2300', ...]
  const entries = [];
  for (let i = 0; i < raw.length; i += 2) {
    entries.push({
      rank: start + (i / 2) + 1,
      username: raw[i],
      score: parseInt(raw[i + 1]),
    });
  }

  const total = await redis.zcard('leaderboard:global');
  return { entries, total, totalPages: Math.ceil(total / pageSize) };
}

14.4 Cache-Aside Pattern — Pattern Chính

Cache-Aside (còn gọi là Lazy Loading) là pattern cache phổ biến nhất. Application tự quản lý cache: check cache trước, nếu miss thì query DB và lưu vào cache. 

JavaScriptCache-aside implementation
const redis = require('./lib/redis');
const User = require('./models/User'); // Mongoose hoặc Prisma

// ── Lấy user với cache ──
async function getUserById(id) {
  const cacheKey = `user:${id}`;

  // 1. Kiểm tra cache
  const cached = await redis.get(cacheKey);
  if (cached) {
    return JSON.parse(cached); // Cache HIT — trả về ngay
  }

  // 2. Cache MISS — query DB
  const user = await User.findById(id).select('-password').lean();
  if (!user) return null;

  // 3. Lưu vào cache với TTL 1 giờ
  await redis.set(cacheKey, JSON.stringify(user), 'EX', 3600);

  return user;
}

// ── Update user — phải invalidate cache ──
async function updateUser(id, data) {
  const user = await User.findByIdAndUpdate(id, data, { new: true }).select('-password');
  if (!user) return null;

  // Xóa cache cũ để lần sau sẽ lấy fresh data từ DB
  await redis.del(`user:${id}`);

  return user;
}

// ── Xóa user — phải xóa cache ──
async function deleteUser(id) {
  await User.findByIdAndDelete(id);
  await redis.del(`user:${id}`);
}

// ── Cache danh sách (phức tạp hơn) ──
async function getProductList({ page = 1, limit = 10, categoryId }) {
  // Tạo cache key từ các params
  const cacheKey = `products:page:${page}:limit:${limit}:cat:${categoryId || 'all'}`;

  const cached = await redis.get(cacheKey);
  if (cached) return JSON.parse(cached);

  // Query DB
  const result = await getProductsFromDB({ page, limit, categoryId });

  // Cache danh sách với TTL ngắn hơn (5 phút) — thay đổi thường xuyên hơn
  await redis.set(cacheKey, JSON.stringify(result), 'EX', 300);

  return result;
}

// Khi có sản phẩm mới/bị xóa — xóa tất cả cache liên quan
async function invalidateProductCache() {
  const keys = await redis.keys('products:page:*');
  if (keys.length > 0) {
    await redis.del(...keys); // Xóa tất cả
  }
}
⚠️ Cache Stampede (Thundering Herd)

Khi cache expire, nhiều request đến cùng lúc đều query DB → DB bị quá tải. Giải pháp: Stale-While-Revalidate (trả về data cũ trong khi refresh) hoặc Mutex Lock (chỉ 1 request query DB, các request khác chờ cache).

14.5 Cache Middleware cho Express

JavaScriptmiddleware/cache.js
// middleware/cache.js
const redis = require('../lib/redis');

/**
 * Cache middleware cho GET requests
 * @param {number} ttl - Thời gian cache tính bằng giây (default: 300 = 5 phút)
 * @param {function} keyFn - Function tạo cache key (optional)
 */
function cacheMiddleware(ttl = 300, keyFn = null) {
  return async (req, res, next) => {
    // Chỉ cache GET requests
    if (req.method !== 'GET') return next();

    // Tạo cache key (có thể include user ID nếu cần per-user cache)
    const cacheKey = keyFn
      ? keyFn(req)
      : `http:${req.path}:${JSON.stringify(req.query)}`;

    try {
      // Kiểm tra cache
      const cached = await redis.get(cacheKey);

      if (cached) {
        // Cache HIT
        const data = JSON.parse(cached);
        res.setHeader('X-Cache', 'HIT');
        res.setHeader('X-Cache-TTL', await redis.ttl(cacheKey));
        return res.json(data);
      }

      // Cache MISS — ghi đè res.json để intercept response
      res.setHeader('X-Cache', 'MISS');

      const originalJson = res.json.bind(res);
      res.json = async (data) => {
        // Chỉ cache response thành công (status 2xx)
        if (res.statusCode >= 200 && res.statusCode < 300) {
          await redis.set(cacheKey, JSON.stringify(data), 'EX', ttl);
        }
        return originalJson(data);
      };

      next();

    } catch (err) {
      // Nếu Redis lỗi, vẫn cho request đi qua (graceful degradation)
      console.error('Cache middleware error:', err.message);
      next();
    }
  };
}

module.exports = { cacheMiddleware };
JavaScriptSử dụng cache middleware
const express = require('express');
const { cacheMiddleware } = require('./middleware/cache');
const redis = require('./lib/redis');

  res.json(products);
});

// Cache 1 giờ cho chi tiết sản phẩm (ít thay đổi)

  const product = await getProductById(req.params.id);
  if (!product) return res.status(404).json({ error: 'Không tìm thấy' });
  res.json(product);
});

// Khi update/delete — xóa cache liên quan
router.put('/products/:id', async (req, res) => {
  const product = await updateProduct(req.params.id, req.body);
  // Invalidate cache
  await redis.del(`product:${req.params.id}`);
  await redis.del(`http:/products:${JSON.stringify({})}`);

});

14.6 Session Store với Redis

Mặc định Express lưu sessions trong MemoryStore — bộ nhớ của Node process. Khi restart server, tất cả sessions mất. Khi chạy nhiều server instances, sessions không được chia sẻ. Redis giải quyết cả 2 vấn đề này. 

ShellTerminal
npm install express-session connect-redis
JavaScriptapp.js — Session với Redis
const express = require('express');
const session = require('express-session');
const { createClient } = require('redis');
const RedisStore = require('connect-redis').default;

const app = express();

// Tạo Redis client riêng cho session store
const redisClient = createClient({
  url: process.env.REDIS_URL,
});
redisClient.connect().catch(console.error);

// Cấu hình session với Redis store
app.use(session({
  store: new RedisStore({
    client: redisClient,
    prefix: 'sess:',          // Key prefix trong Redis
    ttl: 86400,               // Session expire sau 24 giờ (giây)
  }),
  secret: process.env.SESSION_SECRET,
  resave: false,              // Không lưu lại nếu session không thay đổi
  saveUninitialized: false,   // Không tạo session trống
  cookie: {
    secure: process.env.NODE_ENV === 'production', // Chỉ HTTPS trong prod
    httpOnly: true,
    maxAge: 24 * 60 * 60 * 1000, // 24 giờ (milliseconds)
    sameSite: 'strict',
  },
  name: 'sid',               // Tên cookie (thay mặc định 'connect.sid')
}));

// Sử dụng session
app.post('/login', async (req, res) => {
  const user = await authenticateUser(req.body.email, req.body.password);
  if (!user) return res.status(401).json({ error: 'Sai thông tin đăng nhập' });

  // Lưu thông tin vào session (tự động persist vào Redis)
  req.session.userId = user.id;
  req.session.role = user.role;

  // Regenerate session ID để tránh Session Fixation attack
  req.session.regenerate((err) => {
    if (err) return res.status(500).json({ error: 'Session error' });
    req.session.userId = user.id;
    req.session.role = user.role;
    res.json({ message: 'Đăng nhập thành công', user: { id: user.id, name: user.name } });
  });
});

app.post('/logout', (req, res) => {
  req.session.destroy((err) => {
    res.clearCookie('sid');
    res.json({ message: 'Đã đăng xuất' });
  });
});

14.7 Rate Limiting với Redis

Rate limiting giới hạn số request mỗi user/IP có thể thực hiện trong khoảng thời gian. Bảo vệ API khỏi brute force, DoS, và abuse. 

JavaScriptmiddleware/rateLimiter.js
// middleware/rateLimiter.js
const redis = require('../lib/redis');

/**
 * Rate limiter dùng Fixed Window Counter
 * @param {number} maxRequests - Số request tối đa trong window
 * @param {number} windowSecs - Thời gian window (giây)
 * @param {string} prefix - Prefix cho Redis key
 */
function rateLimiter(maxRequests = 100, windowSecs = 60, prefix = 'rl') {
  return async (req, res, next) => {
    // Key theo IP (hoặc user ID nếu authenticated)
    const identifier = req.user?.id || req.ip;
    const key = `${prefix}:${identifier}`;

    try {
      // Pipeline: gộp 2 lệnh vào 1 round-trip
      const pipeline = redis.pipeline();
      pipeline.incr(key);          // Tăng counter
      pipeline.ttl(key);           // Lấy TTL hiện tại
      const results = await pipeline.exec();

      const count = results[0][1];  // Kết quả INCR
      const ttl = results[1][1];    // Kết quả TTL

      // Nếu key mới (TTL = -1), đặt expire
      if (ttl === -1) {
        await redis.expire(key, windowSecs);
      }

      // Set headers để client biết limit
      res.setHeader('X-RateLimit-Limit', maxRequests);
      res.setHeader('X-RateLimit-Remaining', Math.max(0, maxRequests - count));
      res.setHeader('X-RateLimit-Reset', Math.ceil(Date.now() / 1000) + (ttl > 0 ? ttl : windowSecs));

      if (count > maxRequests) {
        return res.status(429).json({
          error: 'Quá nhiều request. Vui lòng thử lại sau.',
          retryAfter: ttl > 0 ? ttl : windowSecs,
        });
      }

      next();

    } catch (err) {
      // Nếu Redis lỗi, cho request đi qua (tránh block toàn bộ service)
      console.error('Rate limiter error:', err.message);
      next();
    }
  };
}

/**
 * Sliding Window Rate Limiter — chính xác hơn Fixed Window
 */
function slidingWindowRateLimiter(maxRequests = 100, windowSecs = 60) {
  return async (req, res, next) => {
    const identifier = req.user?.id || req.ip;
    const key = `sw:rl:${identifier}`;
    const now = Date.now();
    const windowStart = now - windowSecs * 1000;

    try {
      const pipeline = redis.pipeline();
      // Xóa các timestamps cũ hơn window
      pipeline.zremrangebyscore(key, 0, windowStart);
      // Đếm requests trong window
      pipeline.zcard(key);
      // Thêm timestamp hiện tại
      pipeline.zadd(key, now, `${now}-${Math.random()}`);
      // Set TTL
      pipeline.expire(key, windowSecs);

      const results = await pipeline.exec();
      const requestCount = results[1][1]; // ZCARD result

      res.setHeader('X-RateLimit-Limit', maxRequests);
      res.setHeader('X-RateLimit-Remaining', Math.max(0, maxRequests - requestCount - 1));

      if (requestCount >= maxRequests) {
        return res.status(429).json({
          error: 'Rate limit exceeded',
          retryAfter: windowSecs,
        });
      }

      next();
    } catch (err) {
      console.error('Sliding window rate limiter error:', err.message);
      next();
    }
  };
}

module.exports = { rateLimiter, slidingWindowRateLimiter };
JavaScriptÁp dụng rate limiter

14.8 Pub/Sub — Real-Time Events

Redis Pub/Sub cho phép một publisher gửi message đến nhiều subscribers. Dùng để broadcast events khi có thay đổi — ví dụ: thông báo tất cả server instances invalidate cache khi có update.

📌 Quan trọng: Cần 2 Redis Connections

Khi một Redis client đang ở chế độ subscribe, nó chỉ được dùng cho pub/sub, không thể dùng cho các lệnh khác. Vì vậy cần tạo 2 clients riêng: 1 cho pub/sub subscriber, 1 cho tất cả operations còn lại. 

JavaScriptlib/pubsub.js
// lib/pubsub.js
const Redis = require('ioredis');

// Client riêng cho subscriber (không dùng cho operations khác)
const subscriber = new Redis(process.env.REDIS_URL);
const publisher = new Redis(process.env.REDIS_URL);

// Subscribe vào channel
subscriber.subscribe('cache:invalidate', 'user:update', 'order:status', (err, count) => {
  if (err) console.error('Subscribe error:', err);
  else console.log(`📡 Subscribed to ${count} channels`);
});

// Xử lý messages nhận được
subscriber.on('message', async (channel, message) => {
  console.log(`📨 [${channel}]:`, message);

  if (channel === 'cache:invalidate') {
    const { patterns } = JSON.parse(message);
    // Xóa cache keys match pattern
    for (const pattern of patterns) {
      const keys = await redis.keys(pattern);
      if (keys.length > 0) await redis.del(...keys);
    }
  }

  if (channel === 'user:update') {
    const { userId } = JSON.parse(message);
    await redis.del(`user:${userId}`); // Invalidate user cache
  }
});

// Publisher
async function publishCacheInvalidation(patterns) {
  await publisher.publish('cache:invalidate', JSON.stringify({ patterns }));
}

async function publishUserUpdate(userId) {
  await publisher.publish('user:update', JSON.stringify({ userId }));
}

async function publishOrderStatus(orderId, status) {
  await publisher.publish('order:status', JSON.stringify({ orderId, status }));
}

module.exports = { publishCacheInvalidation, publishUserUpdate, publishOrderStatus };

14.9 Distributed Lock — Tránh Race Condition

Khi nhiều server instances cùng xử lý một job, race condition có thể xảy ra. Distributed lock đảm bảo chỉ 1 instance thực sự xử lý tại một thời điểm. 

JavaScriptlib/lock.js
// lib/lock.js — Distributed Lock với Redis
const redis = require('./redis');
const { randomBytes } = require('crypto');

/**
 * Thử lấy lock
 * @param {string} resource - Tên resource cần lock (vd: 'order:123')
 * @param {number} ttl - Thời gian lock tối đa (ms)
 * @returns {string|null} - lockId nếu thành công, null nếu đang bị lock
 */
async function acquireLock(resource, ttl = 10000) {
  const lockKey = `lock:${resource}`;
  const lockId = randomBytes(16).toString('hex'); // Unique ID để xác nhận owner

  // SET key value NX PX ttl
  // NX = chỉ set nếu chưa tồn tại
  // PX = expire tính bằng milliseconds
  const result = await redis.set(lockKey, lockId, 'PX', ttl, 'NX');

  if (result === 'OK') {
    return lockId; // Lấy lock thành công
  }
  return null; // Lock đã bị giữ bởi instance khác
}

/**
 * Giải phóng lock (chỉ nếu là owner)
 */
async function releaseLock(resource, lockId) {
  const lockKey = `lock:${resource}`;

  // Dùng Lua script để đảm bảo atomic: kiểm tra owner và xóa trong 1 bước
  const luaScript = `
    if redis.call("get", KEYS[1]) == ARGV[1] then
      return redis.call("del", KEYS[1])
    else
      return 0
    end
  `;

  const result = await redis.eval(luaScript, 1, lockKey, lockId);
  return result === 1;
}

/**
 * Wrapper tiện lợi — tự acquire và release
 */
async function withLock(resource, ttl, fn) {
  const lockId = await acquireLock(resource, ttl);

  if (!lockId) {
    throw new Error(`Resource "${resource}" đang bị khóa. Thử lại sau.`);
  }

  try {
    return await fn();
  } finally {
    await releaseLock(resource, lockId);
  }
}

module.exports = { acquireLock, releaseLock, withLock };
JavaScriptSử dụng distributed lock
const { withLock } = require('./lib/lock');

// Xử lý payment — chỉ 1 instance được chạy cùng lúc cho cùng orderId

  const { id: orderId } = req.params;

  try {
    const result = await withLock(`payment:${orderId}`, 30000, async () => {
      // Kiểm tra order chưa được thanh toán
      const order = await getOrder(orderId);

        throw new Error('Order đã được xử lý');
      }

      // Xử lý thanh toán

      // Update order status
      await updateOrderStatus(orderId, 'PAID');

      return payment;
    });

    res.json({ success: true, payment: result });

  } catch (error) {
    if (error.message.includes('đang bị khóa')) {
      return res.status(409).json({ error: 'Đang xử lý, thử lại sau' });
    }
    res.status(400).json({ error: error.message });
  }
});
